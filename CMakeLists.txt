cmake_minimum_required(VERSION 3.20)
project(Poppy VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(poppy_core src/text.cpp src/diff.cpp src/terminal_model.cpp src/tree_model.cpp src/session.cpp src/file_search.cpp)
target_include_directories(poppy_core PUBLIC src)
if(MSVC)
  target_compile_options(poppy_core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(poppy_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Prefer the official amalgamation so Windows releases keep a single exe.
# Native core tests can still run before the dependency has been supplied.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.c" AND
   EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.h")
  enable_language(C)
  add_library(poppy_sqlite STATIC third_party/sqlite/sqlite3.c)
  target_include_directories(poppy_sqlite PUBLIC third_party/sqlite)
  target_compile_definitions(poppy_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION)
  if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(poppy_sqlite PRIVATE Threads::Threads)
  endif()
  if(MSVC)
    set_property(TARGET poppy_sqlite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  set(POPPY_SQLITE_TARGET poppy_sqlite)
else()
  find_package(SQLite3 QUIET)
  if(SQLite3_FOUND)
    set(POPPY_SQLITE_TARGET SQLite::SQLite3)
  elseif(WIN32)
    message(FATAL_ERROR "SQLite3 is required. Place sqlite3.c and sqlite3.h from the official SQLite amalgamation in third_party/sqlite/.")
  else()
    message(STATUS "SQLite3 not available: building core tests only. Add third_party/sqlite/sqlite3.c and sqlite3.h to enable storage tests.")
  endif()
endif()
if(POPPY_SQLITE_TARGET)
  add_library(poppy_storage src/storage.cpp)
  target_include_directories(poppy_storage PUBLIC src)
  target_link_libraries(poppy_storage PUBLIC poppy_core PRIVATE ${POPPY_SQLITE_TARGET})
  if(MSVC)
    target_compile_options(poppy_storage PRIVATE /W4 /utf-8 /permissive-)
    set_property(TARGET poppy_storage PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(poppy_storage PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(WIN32)
  enable_language(RC)
  add_executable(Poppy WIN32 src/app.cpp src/viewer.cpp src/terminal.cpp src/win.cpp src/render.cpp src/tree.cpp src/activity_button.cpp src/scrollbars.cpp src/quick_open.cpp src/Poppy.rc)
  target_include_directories(Poppy PRIVATE src)
  target_link_libraries(Poppy PRIVATE poppy_core poppy_storage comctl32 shell32 ole32 user32 d2d1 dwrite dwmapi imm32 uuid)
  target_compile_definitions(Poppy PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000006)
  if(MSVC)
    target_compile_options(Poppy PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive->")
    target_link_options(Poppy PRIVATE /MANIFEST:NO)
    set_property(TARGET Poppy poppy_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_compile_options(Poppy PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-pthread>")
    # Unicode GUI entry point; link the C++ and winpthreads runtimes into the exe.
    target_link_options(Poppy PRIVATE -municode -static -pthread)
  endif()
endif()

include(CTest)

  add_executable(poppy_core_tests tests/core_tests.cpp)

  if(MSVC)
    target_compile_options(poppy_core_tests PRIVATE /utf-8)
    set_property(TARGET poppy_core_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  add_test(NAME poppy_core_tests COMMAND poppy_core_tests)
  add_executable(poppy_diff_tests tests/diff_tests.cpp)
  target_link_libraries(poppy_diff_tests PRIVATE poppy_core)
  if(MSVC)
    target_compile_options(poppy_diff_tests PRIVATE /W4 /utf-8 /permissive-)
    set_property(TARGET poppy_diff_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(poppy_diff_tests PRIVATE -Wall -Wextra -Wpedantic)
    if(MINGW)
      target_link_options(poppy_diff_tests PRIVATE -static -pthread)
    endif()
  endif()
  add_test(NAME poppy_diff_tests COMMAND poppy_diff_tests)
  add_executable(poppy_tree_tests tests/tree_tests.cpp)
  target_link_libraries(poppy_tree_tests PRIVATE poppy_core)
  if(MSVC)
    target_compile_options(poppy_tree_tests PRIVATE /W4 /utf-8 /permissive-)
    set_property(TARGET poppy_tree_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(poppy_tree_tests PRIVATE -Wall -Wextra -Wpedantic)
    if(MINGW)
      target_link_options(poppy_tree_tests PRIVATE -static -pthread)
    endif()
  endif()
  add_test(NAME poppy_tree_tests COMMAND poppy_tree_tests)
  if(WIN32)
    add_executable(poppy_render_tests tests/render_tests.cpp src/render.cpp src/scrollbars.cpp)
    target_compile_definitions(poppy_render_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(poppy_render_tests PRIVATE poppy_core d2d1 dwrite windowscodecs user32 ole32 uuid)
    if(MSVC)
      target_compile_options(poppy_render_tests PRIVATE /W4 /utf-8 /permissive-)
      set_property(TARGET poppy_render_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
      target_compile_options(poppy_render_tests PRIVATE -Wall -Wextra -Wpedantic)
      target_link_options(poppy_render_tests PRIVATE -static -pthread)
    endif()
    add_test(NAME poppy_render_tests COMMAND poppy_render_tests)
    set_tests_properties(poppy_render_tests PROPERTIES TIMEOUT 30)
    add_executable(poppy_folder_tests tests/folder_tests.cpp)
    target_include_directories(poppy_folder_tests PRIVATE src)
    target_compile_definitions(poppy_folder_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(poppy_folder_tests PRIVATE user32)
    if(MSVC)
      target_compile_options(poppy_folder_tests PRIVATE /W4 /utf-8 /permissive-)
      set_property(TARGET poppy_folder_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
      target_compile_options(poppy_folder_tests PRIVATE -Wall -Wextra -Wpedantic)
      target_link_options(poppy_folder_tests PRIVATE -municode -static -pthread)
    endif()
    add_dependencies(poppy_folder_tests Poppy)
    add_test(NAME poppy_folder_tests COMMAND poppy_folder_tests $<TARGET_FILE:Poppy>)
    set_tests_properties(poppy_folder_tests PROPERTIES TIMEOUT 60)
  endif()
endif()
