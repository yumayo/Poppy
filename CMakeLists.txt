# Prefer the official amalgamation so the application DLL needs no SQLite DLL.
# Native core tests can still run before the dependency has been supplied.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.c" AND
   EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.h")
  enable_language(C)
  add_library(poppy_sqlite STATIC third_party/sqlite/sqlite3.c)
  target_include_directories(poppy_sqlite PUBLIC third_party/sqlite)
  target_compile_definitions(poppy_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION)
  if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(poppy_sqlite PRIVATE Threads::Threads)
  endif()
  if(MSVC)
    set_property(TARGET poppy_sqlite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  set(POPPY_SQLITE_TARGET poppy_sqlite)
else()
  find_package(SQLite3 QUIET)
  if(SQLite3_FOUND)
    set(POPPY_SQLITE_TARGET SQLite::SQLite3)
  elseif(WIN32)
    message(FATAL_ERROR "SQLite3 is required. Place sqlite3.c and sqlite3.h from the official SQLite amalgamation in third_party/sqlite/.")
  else()
    message(STATUS "SQLite3 not available: building core tests only. Add third_party/sqlite/sqlite3.c and sqlite3.h to enable storage tests.")
  endif()
endif()

    message(FATAL_ERROR "WebView2 SDK is required. Unpack Microsoft.Web.WebView2 1.0.4314.35 into third_party/microsoft.web.webview2.1.0.4314.35 or set POPPY_WEBVIEW2_SDK.")
  endif()
  function(poppy_add_webview2 target)
    target_sources(${target} PRIVATE src/html_preview_win.cpp)
    target_include_directories(${target} SYSTEM PRIVATE "${POPPY_WEBVIEW2_SDK}/build/native/include")
    if(MINGW)
      target_include_directories(${target} PRIVATE src/webview2_mingw)
    endif()
    target_link_libraries(${target} PRIVATE shell32 urlmon)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${POPPY_WEBVIEW2_LOADER}" "$<TARGET_FILE_DIR:${target}>/WebView2Loader.dll"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${POPPY_WEBVIEW2_SDK}/LICENSE.txt" "$<TARGET_FILE_DIR:${target}>/WebView2-LICENSE.txt"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${POPPY_WEBVIEW2_SDK}/NOTICE.txt" "$<TARGET_FILE_DIR:${target}>/WebView2-NOTICE.txt")
  endfunction()

  # safe while that installation is running and never replaces its Poppy.exe.

    COMMAND ${CMAKE_COMMAND}
      "-DOUTPUT_DIR=${POPPY_PUBLISH_DIR}"

      "-DWSL_HELPER=${POPPY_WSL_HELPER}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/publish.cmake"
    DEPENDS Poppy poppy_app
    VERBATIM)
endif()
