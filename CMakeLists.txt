cmake_minimum_required(VERSION 3.20)
project(Poppy VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)

# Prefer the official amalgamation so Windows releases keep a single exe.
# Native core tests can still run before the dependency has been supplied.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.c" AND
   EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.h")
  enable_language(C)
  add_library(poppy_sqlite STATIC third_party/sqlite/sqlite3.c)
  target_include_directories(poppy_sqlite PUBLIC third_party/sqlite)
  target_compile_definitions(poppy_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION)
  if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(poppy_sqlite PRIVATE Threads::Threads)
  endif()
  if(MSVC)
    set_property(TARGET poppy_sqlite PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  set(POPPY_SQLITE_TARGET poppy_sqlite)
else()
  find_package(SQLite3 QUIET)
  if(SQLite3_FOUND)
    set(POPPY_SQLITE_TARGET SQLite::SQLite3)
  elseif(WIN32)
    message(FATAL_ERROR "SQLite3 is required. Place sqlite3.c and sqlite3.h from the official SQLite amalgamation in third_party/sqlite/.")
  else()
    message(STATUS "SQLite3 not available: building core tests only. Add third_party/sqlite/sqlite3.c and sqlite3.h to enable storage tests.")
  endif()
endif()

if(WIN32)
  enable_language(RC)
  add_executable(Poppy WIN32 src/workspace.cpp src/app.cpp src/project_provider.cpp src/wsl_client.cpp src/wsl_dialog.cpp src/ai_status_win.cpp src/hook_pipe.cpp src/terminal_pane.cpp src/viewer.cpp src/find_bar.cpp src/search_pane.cpp src/terminal.cpp src/win.cpp src/render.cpp src/tree.cpp src/activity_button.cpp src/scrollbars.cpp src/quick_open.cpp src/Poppy.rc)
  set(POPPY_WSL_HELPER "" CACHE FILEPATH "Linux x86_64 helper to bundle beside Poppy.exe")
  if(POPPY_WSL_HELPER)
    if(NOT EXISTS "${POPPY_WSL_HELPER}")
      message(FATAL_ERROR "POPPY_WSL_HELPER does not exist: ${POPPY_WSL_HELPER}")
    endif()
    set_property(TARGET Poppy APPEND PROPERTY LINK_DEPENDS "${POPPY_WSL_HELPER}")
    add_custom_command(TARGET Poppy POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:Poppy>/wsl"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${POPPY_WSL_HELPER}" "$<TARGET_FILE_DIR:Poppy>/wsl/poppy-wsl-helper")
  endif()
  add_custom_command(TARGET Poppy POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E rm -rf "$<TARGET_FILE_DIR:Poppy>/ai"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/integration/ai" "$<TARGET_FILE_DIR:Poppy>/ai")
  target_include_directories(Poppy PRIVATE src)
  target_sources(Poppy PRIVATE src/clipboard_image_win.cpp src/image_preview_win.cpp)
  target_sources(Poppy PRIVATE src/single_instance.cpp)
  target_link_libraries(Poppy PRIVATE windowscodecs)
  target_link_libraries(Poppy PRIVATE poppy_core poppy_storage comctl32 shell32 ole32 user32 d2d1 dwrite dwmapi imm32 uuid)
  target_compile_definitions(Poppy PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000006)
  if(MSVC)
    target_compile_options(Poppy PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive->")
    target_link_options(Poppy PRIVATE /MANIFEST:NO)
    set_property(TARGET Poppy poppy_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_compile_options(Poppy PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-pthread>")
    # Unicode GUI entry point; link the C++ and winpthreads runtimes into the exe.
    target_link_options(Poppy PRIVATE -municode -static -pthread)
  endif()
endif()
